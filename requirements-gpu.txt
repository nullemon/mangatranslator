# ── Optional GPU acceleration (recommended on your RTX 4080) ──
# Unlocks the trained speech-balloon segmentation model, which produces precise
# per-balloon pixel masks — far better text removal and containment than CV.
# Without these, the app automatically falls back to the built-in CV detector.
#
# Install with:  ./setup_gpu.sh
#
# On Linux, the default torch wheels already ship CUDA support for the 4080.
torch>=2.2.0
torchvision>=0.17.0
ultralytics>=8.2.0
huggingface_hub[hf_xet]>=0.24.0

# ── Local OCR + clean removal (all run on your GPU, all free) ──
# manga-ocr: reads each bubble's own Japanese text so a translation can never
#   land in the wrong bubble (and never misses a detected bubble).
# simple-lama-inpainting: GPU LaMa model for clean text erasure over artwork
#   and screentones (much better than OpenCV inpaint).
manga-ocr>=0.1.11
simple-lama-inpainting>=0.1.0

# ── Free text detection (narration, labels, dramatic text) ──
# CRAFT finds text anywhere on the page, not just inside bubbles.
# NOTE: craft-text-detector itself is deliberately NOT listed here — its
# setup pins an ancient opencv (<4.5.4.62) that no longer builds on modern
# Python and would break this whole file's install. setup_gpu.sh installs it
# with --no-deps instead. These are the deps it actually needs at runtime:
scipy>=1.10.0
gdown>=4.6.0

# ── Text-pixel segmentation (comic-text-detector, ONNX) ──
# Marks every text stroke on the page at pixel level — manga-trained, so
# erasure is surgical: whole characters removed, art never touched.
# Weights auto-download to models/comictextdetector.pt.onnx on first run.
onnxruntime-gpu>=1.17.0

# ── Super-resolution (Real-ESRGAN anime, via spandrel) ──
# Upscales low-res raws before the pipeline runs: sharper OCR, cleaner
# erasure edges, crisper lettering. Weights auto-download (~18MB).
spandrel>=0.3.0
